<?php
include('db.php');
session_start();
$message = "";

if (isset($_POST['login'])) {
    $username = strtoupper(trim($_POST['username'] ?? ''));
    $password = trim($_POST['password'] ?? '');

    $stmt = mysqli_prepare($conn, "SELECT * FROM users WHERE username = ?");
    mysqli_stmt_bind_param($stmt, "s", $username);
    mysqli_stmt_execute($stmt);
    $query = mysqli_stmt_get_result($stmt);
    
    if ($query && mysqli_num_rows($query) > 0) {
        $user = mysqli_fetch_assoc($query);
        
        if (md5($password) === $user['password']) {
            
            if ($user['username'] === 'GODFREY') {
                $_SESSION['admin'] = $user['username'];
                header('Location: admin_dashboard.php');
                exit();
            }
            
            if (isset($user['status']) && $user['status'] == 'pending') {
                $message = "<div class='alert alert-warning text-center fw-bold py-2 mb-3' style='font-size: 13px;'>⚠️ you cant login yet. your account is not approved by the Admin.</div>";
            } else {
                $_SESSION['user'] = $user['username'];
                header('Location: index.php'); 
                exit();
            }
            
        } else {
            $message = "<div class='alert alert-danger text-center fw-bold py-2 mb-3' style='font-size: 13px;'>Wrong password!</div>";
        }
    } else {
        $message = "<div class='alert alert-danger text-center fw-bold py-2 mb-3' style='font-size: 13px;'>Username not found!</div>";
    }
}

if (isset($_POST['reset_password'])) {
    $username = strtoupper(trim($_POST['hidden_fp_username'] ?? ''));
    $otp_code = trim($_POST['fp_otp'] ?? '');
    $new_password = $_POST['fp_new_password'] ?? '';
    $confirm_password = $_POST['fp_confirm_password'] ?? '';

    $stmt = mysqli_prepare($conn, "SELECT * FROM users WHERE username = ?");
    mysqli_stmt_bind_param($stmt, "s", $username);
    mysqli_stmt_execute($stmt);
    $query = mysqli_stmt_get_result($stmt);
    
    if ($query && mysqli_num_rows($query) > 0) {
        $user = mysqli_fetch_assoc($query);
        
        if (empty($user['reset_otp']) || $user['reset_otp'] !== $otp_code) {
            $message = "<div class='alert alert-danger text-center fw-bold py-2 mb-3' style='font-size: 13px;'>The OTP code you entered is incorrect!</div>";
        } elseif (strtotime(date("Y-m-d H:i:s")) > strtotime($user['otp_expiry'])) {
            $message = "<div class='alert alert-danger text-center fw-bold py-2 mb-3' style='font-size: 13px;'>The OTP has expired. Please request a new one.</div>";
        } else {
            if (strlen($new_password) < 8 || !preg_match('/[A-Z]/', $new_password) || !preg_match('/[0-9]/', $new_password)) {
                $message = "<div class='alert alert-danger text-center fw-bold py-2 mb-3' style='font-size: 13px;'>Password must be at least 8 characters long, contain at least 1 uppercase letter, and numbers!</div>";
            } elseif ($new_password === $confirm_password) {
                $final_password = md5($new_password);
                
                $update_stmt = mysqli_prepare($conn, "UPDATE users SET password = ?, reset_otp = NULL, otp_expiry = NULL WHERE username = ?");
                mysqli_stmt_bind_param($update_stmt, "ss", $final_password, $username);
                $update = mysqli_stmt_execute($update_stmt);
                
                if ($update) {
                    $message = "<div class='alert alert-success text-center fw-bold py-2 mb-3' style='font-size: 13px;'>Password successfully reset! You can now log in.</div>";
                } else {
                    $message = "<div class='alert alert-danger text-center fw-bold py-2 mb-3' style='font-size: 13px;'>Database error. Please try again.</div>";
                }
            } else {
                $message = "<div class='alert alert-danger text-center fw-bold py-2 mb-3' style='font-size: 13px;'>New passwords do not match!</div>";
            }
        }
    } else {
        $message = "<div class='alert alert-danger text-center fw-bold py-2 mb-3' style='font-size: 13px;'>Username not found!</div>";
    }
}

    <script>
        function togglePassword(fieldId, checkboxId) {
            var pwdInput = document.getElementById(fieldId);
            var checkbox = document.getElementById(checkboxId);
            if (checkbox.checked) {
                pwdInput.type = "text";
            } else {
                pwdInput.type = "password";
            }
        }

        function toggleMultiplePasswords(fieldIds, checkboxId) {
            var checkbox = document.getElementById(checkboxId);
            fieldIds.forEach(function(id) {
                var pwdInput = document.getElementById(id);
                if (checkbox.checked) {
                    pwdInput.type = "text";
                } else {
                    pwdInput.type = "password";
                }
            });
        }

        document.getElementById('sendOtpBtn').addEventListener('click', function() {
            let username = document.getElementById('fp_username').value;
            let email = document.getElementById('fp_email').value;
            let msgBox = document.getElementById('otp-response-msg');

            if (!username || !email) {
                msgBox.innerHTML = '<div class="alert alert-danger py-1 small mb-2">PLEASE INPUT USERNAME AND EMAIL!</div>';
                return;
            }

            let formData = new FormData();
            formData.append('username', username);
            formData.append('email', email);

            fetch('send_otp.php', {
                method: 'POST',
                body: formData
            })
            .then(response => response.json())
            .then(data => {
                if(data.status === 'success') {
                    msgBox.innerHTML = `<div class="alert alert-success py-1 small mb-2">${data.message}</div>`;
                    
                    document.getElementById('hidden_fp_username').value = username;

                    document.getElementById('step-email-request').style.display = 'none';
                    document.getElementById('step-reset-form').style.display = 'block';
                    document.getElementById('resetBtn').style.display = 'block';
                } else {
                    msgBox.innerHTML = `<div class="alert alert-danger py-1 small mb-2">${data.message}</div>`;
                }
            })
            .catch(error => {
                msgBox.innerHTML = '<div class="alert alert-danger py-1 small mb-2">There was an error sending the OTP.</div>';
            });
        });
    </script>
